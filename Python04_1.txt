#criando a lista com 5 coisas que gosto
>>> lista = ['manga', 'melancia', 'jogos', 'LEGO', 'cachorro']

#imprimindo a lista
>>> print(lista)
['manga', 'melancia', 'jogos', 'LEGO', 'cachorro']

#imprimindo o elemento do meio 
>>> print(lista[2])
jogos

#subistituindo o elemento do meio
>>> lista[2] = 'canario'

#imprimindo a nova lista
>>> print(lista)
['manga', 'melancia', 'canario', 'LEGO', 'cachorro']

#adicionando um novo elemento no final
>>> lista.append('doce')
>>> print(lista)
['manga', 'melancia', 'canario', 'LEGO', 'cachorro', 'doce']

#adicionando um novo elemento no começo
>>> lista.insert(0, 'filme')
>>> print(lista)
['filme', 'manga', 'melancia', 'canario', 'LEGO', 'cachorro', 'doce']

#adicionando um novo elemento na terceira 
>>> lista.insert(2, 'marvel')
>>> print(lista)
['filme', 'manga', 'marvel', 'melancia', 'canario', 'LEGO', 'cachorro', 'doce']

#removendo o ultimo elemento
>>> lista.pop()
'doce'
>>> print(lista)
['filme', 'manga', 'marvel', 'melancia', 'canario', 'LEGO', 'cachorro']

#removendo o primeiro elemento
>>> lista.pop(0)
'filme'
>>> print(lista)
['manga', 'marvel', 'melancia', 'canario', 'LEGO', 'cachorro']

#removendo o elemento na segunda posiçao
>>> lista.pop(1)
'marvel'
>>> print(lista)
['manga', 'melancia', 'canario', 'LEGO', 'cachorro']

#criando uma string a partir da lista com join
>>> string =','.join(lista)
>>> print(string)
manga,melancia,canario,LEGO,cachorro
